# Carino Hyprland -- runtime packages, as Fedora names (one per line, "# purpose").
# setup.sh derives its hypr_<family> arrays from this list.
# Hyprland itself must be >= 0.55 (Lua config); on Fedora it comes from the
# copr lionheartp/Hyprland.

# --- session ---
hyprland                       # the compositor (reads ~/.config/hypr/hyprland.lua)
xdg-desktop-portal-hyprland    # screen sharing, screenshots portal
xdg-desktop-portal-gtk         # file chooser portal (float-portal-file-chooser rule)
hyprpolkitagent                # password prompts for GUI admin actions (autostart)

# --- desktop pieces ---
waybar                         # the bar (needs the ext/workspaces module; tested on 0.15)
hyprpaper                      # wallpaper (block syntax needs hyprpaper >= 0.8)
hyprlock                       # lock screen (SUPER+ALT+L, hypridle)
hypridle                       # idle lock and screens off
mako                           # notifications, makoctl for the notification keys
wofi                           # launcher, window switcher, clipboard picker, power menu
kitty                          # terminal (ALT+Q, SUPER+Return drop-down)
nautilus                       # file manager behind xdg-open (ALT+W)
xdg-utils                      # xdg-open
pavucontrol                    # volume mixer (SUPER+A, bar volume click)
htop                           # bar CPU / memory click
NetworkManager-tui             # nmtui, the bar network click
procps-ng                      # pidof / pkill used by autostart and keys

# --- capture / clipboard ---
grim                           # screenshots
slurp                          # region selection for screenshots
wl-clipboard                   # wl-copy / wl-paste
cliphist                       # clipboard history (SUPER+V)
jq                             # reads the focused monitor for ALT+S

# --- audio / media / network ---
wireplumber                    # wpctl for the volume keys and bar clicks
pipewire-pulseaudio            # PulseAudio server for pavucontrol and waybar's pulseaudio module
playerctl                      # media keys
brightnessctl                  # brightness keys

# --- fonts ---
fontconfig                     # fc-cache for the bundled fonts
ibm-plex-sans-fonts            # bar, wofi, mako text
ibm-plex-mono-fonts            # kitty
fontawesome-6-free-fonts       # bar icons (Font Awesome 6 Free Solid)
fontawesome-6-brands-fonts     # bar window-title icons (Font Awesome 6 Brands)
google-noto-color-emoji-fonts  # emoji fallback in bar and wofi

# --- python ---
python3                        # scripts/window-switcher.py (SUPER+W)
